public class CountOpportunityLineItem {

    public static void countOliInAccount(List<OpportunityLineItem> oliList) {

        Set<Id> oppIds = new Set<Id>();
        for (OpportunityLineItem oli : oliList) {
            if (oli.OpportunityId != null) {
                oppIds.add(oli.OpportunityId);
            }
        }
        if (oppIds.isEmpty()) {
            return;
        }

        Set<Id> accIds = new Set<Id>();
        for (Opportunity opp : [SELECT AccountId FROM Opportunity WHERE Id IN :oppIds]) {
            if (opp.AccountId != null) {
                accIds.add(opp.AccountId);
            }
        }
        if (accIds.isEmpty()) {
            return;
        }

        Map<Id, Integer> accCountMap = new Map<Id, Integer>();
        for (AggregateResult ar : [SELECT Opportunity.AccountId accId, COUNT(Id) cnt
                                   FROM OpportunityLineItem
                                   WHERE Opportunity.AccountId IN :accIds
                                   GROUP BY Opportunity.AccountId]) {
            accCountMap.put((Id) ar.get('accId'), (Integer) ar.get('cnt'));
        }

        List<Account> accToUpdate = new List<Account>();
        for (Account acc : [SELECT Id, NumberOfLineItem__c FROM Account WHERE Id IN :accIds]) {
            Integer newCount = accCountMap.containsKey(acc.Id) ? accCountMap.get(acc.Id) : 0;
            if (acc.NumberOfLineItem__c != newCount) {
                acc.NumberOfLineItem__c = newCount;
                accToUpdate.add(acc);
            }
        }

        if (!accToUpdate.isEmpty()) {
            update accToUpdate;
        }
    }
}
